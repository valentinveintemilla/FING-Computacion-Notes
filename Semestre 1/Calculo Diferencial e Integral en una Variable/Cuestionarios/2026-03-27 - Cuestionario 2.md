# Ejercicio 2
$$
\begin{align}
&\int_{2}^{5} f(t) dt = 5 \\
&\int_{2}^{10} f(t) dt = 1 \\
&\int_{5}^{10} g(t) dt = 3 \\
 \\
&\int_{5}^{10} 2g(t) - f(t) dt = \text{?} \\
 \\
&\int_{5}^{10} 2g(t) - f(t) dt = \int_{5}^{10} 2g(t) dt - \int_{5}^{10} f(t) dt \\
 \\
&\int_{5}^{10} 2g(t) dt = 2 \cdot \int_{5}^{10} g(t) dt = 2 \cdot 3 = 6 \\
&\int_{5}^{10} f(t) dt = \int_{2}^{10} f(t) dt - \int_{2}^{5} f(t) dt = 1 - 5 = -4 \\
 \\
&\int_{5}^{10} 2g(t) dt - \int_{5}^{10} f(t) dt = 6 - (-4) = 6 + 4 = 10 \\
&\implies \int_{5}^{10} 2g(t) - f(t) dt = 10
\end{align}
$$
